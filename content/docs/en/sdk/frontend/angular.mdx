# Angular SDK

The `@nexus-signal/angular` package provides reactive Angular bindings for in-app notifications, dynamic guides, and subscriber settings matrices. It requires **Angular 17+** and leverages native Angular Signals.

---

Install the package along with its peer dependencies:

## Workspace Setup

Configure the global provider inside your application configuration:

```ts title="app.config.ts"
import { ApplicationConfig } from '@angular/core';
import { provideNexus } from '@nexus-signal/angular';

export const appConfig: ApplicationConfig = {
  providers: [
    provideNexus({
      publicKey: 'pk_prd_your_public_key', // pk_dev_... or pk_prd_...
      subscriberId: 'user_active_99',
      hmacSignature: 'your_computed_hmac_hex',
      apiBaseUrl: 'https://api.nexussignal.dev',
    }),
  ],
};
```

## Direct Component Selectors

Import component symbols into your standalone components:

```ts title="components/dashboard.component.ts"
import { Component } from '@angular/core';
import {
  NexusNotificationBellComponent,
  NexusNotificationInboxComponent,
  NexusNotificationToastComponent,
} from '@nexus-signal/angular';

@Component({
  selector: 'app-dashboard',
  standalone: true,
  imports: [
    NexusNotificationBellComponent,
    NexusNotificationInboxComponent,
    NexusNotificationToastComponent,
  ],
  template: `
    <!-- Top Nav popover notification bell -->
    <nexus-notification-bell variant="ghost" position="bottom-end" />

    <!-- Full-page inbox feed with tag filter -->
    <nexus-notification-inbox tag="billing" [maxItems]="50" />

    <!-- Toast alert overlays -->
    <nexus-notification-toast position="top-right" />
  `,
})
export class DashboardComponent {}
```

## Headless Services & Signals

Inject services directly for custom template interfaces:

### 1. In-App Notification Service (`NexusNotificationsService`)

```ts title="components/custom-feed.component.ts"
import { Component, inject } from '@angular/core';
import { CommonModule } from '@angular/common';
import { NexusNotificationsService } from '@nexus-signal/angular';

@Component({
  selector: 'app-custom-feed',
  standalone: true,
  imports: [CommonModule],
  template: `
    <div class="header">
      <h4>Notifications ({{ notifications.unread() }} unread)</h4>
      <button (click)="notifications.markAllRead()">Mark all read</button>
    </div>

    <div class="list">
      <div
        *ngFor="let item of notifications.items()"
        [class.unread]="!item.read"
        (click)="item.logId && notifications.markRead(item.logId)"
      >
        <h5>{{ item.title }}</h5>
        <p>{{ item.body }}</p>
      </div>
    </div>

    <button *ngIf="notifications.hasMore()" (click)="notifications.loadMore()">
      {{ notifications.loadingMore() ? 'Loading...' : 'Load More' }}
    </button>
  `,
})
export class CustomFeedComponent {
  readonly notifications = inject(NexusNotificationsService);
}
```

#### Observable Service Signals

| Signal Signature | Return Type           | Description                                              |
| :--------------- | :-------------------- | :------------------------------------------------------- |
| `items()`        | `NexusNotification[]` | Reactive list of loaded notifications.                   |
| `unread()`       | `number`              | Computed counter of unread, active notifications.        |
| `loading()`      | `boolean`             | True if the initial feed request is fetching.            |
| `loadingMore()`  | `boolean`             | True if a pagination request is active.                  |
| `hasMore()`      | `boolean`             | True if additional history is available in the database. |
| `connected()`    | `boolean`             | True if the WebSocket handshake is active.               |

### 2. Guide Interceptor Service (`NexusGuidesService`)

Fetch and dismiss in-app messages programmatically:

```ts
import { Component, inject, OnInit } from '@angular/core';
import { NexusGuidesService } from '@nexus-signal/angular';

@Component({
  selector: 'app-guides',
  standalone: true,
  template: `
    <div *ngFor="let g of guidesService.guides()" class="guide-banner">
      <h6>{{ g.content.title }}</h6>
      <p>{{ g.content.body }}</p>
      <button (click)="guidesService.markSeen(g.id)">Acknowledge</button>
      <button (click)="guidesService.dismiss(g.id)">Close</button>
    </div>
  `,
})
export class GuidesComponent implements OnInit {
  readonly guidesService = inject(NexusGuidesService);

  ngOnInit() {
    // Fetch guides scoped to the current path
    this.guidesService.fetch({ pathname: window.location.pathname });
  }
}
```

### 3. Preferences Service (`NexusPreferencesService`)

Create a customized preference form utilizing Signals:

```ts
import { Component, inject } from '@angular/core';
import { NexusPreferencesService } from '@nexus-signal/angular';

@Component({
  selector: 'app-user-preferences',
  standalone: true,
  template: `
    <div *ngFor="let topic of prefService.topics()">
      <span>{{ topic.displayName }}</span>
      <input
        type="checkbox"
        [checked]="prefService.preferences()[topic.id]?.['EMAIL']"
        (change)="toggleEmail(topic.id, $event)"
      />
    </div>
  `,
})
export class UserPreferencesComponent {
  readonly prefService = inject(NexusPreferencesService);

  toggleEmail(topicId: string, event: Event) {
    const checked = (event.target as HTMLInputElement).checked;
    this.prefService.updatePreference(topicId, 'EMAIL', checked);
  }
}
```

## Styling & Themes

Angular components look for custom CSS custom properties (just like React):

- `--nexus-theme-panel`: Background of bells, dropdowns, and cards.
- `--nexus-theme-text`: Primary text color.
- `--nexus-theme-accent`: Accent highlight color (badge counts).
- `--nexus-theme-border`: Component borders.
